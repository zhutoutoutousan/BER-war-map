% --- Why BER+ ---
\begin{frame}{Four coordination priorities}
  \begin{enumerate}
    \item \textbf{Matching} --- connect members to land, peers, and corridor assets
    \item \textbf{Visibility} --- shared view of gewerbe land, industry, power, transport
    \item \textbf{Member value} --- personalized paths and evidence on the map
    \item \textbf{Alignment} --- one briefing surface for leadership and stakeholders
  \end{enumerate}
  \vspace{0.8em}
  The \textbf{BER+ coordination war room} puts Flughafenregion context on a single interactive map.
\end{frame}

% --- Why now ---
\begin{frame}{Why the corridor, why now}
  \begin{itemize}
    \item Grid congestion and \textbf{Netzanschluss} shape where gewerbe can grow
    \item FBB / FEW energy and resilience targets drive near-term investment choices
    \item \textbf{SEGRO Pilot-1} (2\,ha) anchors Phase I development on the map
    \item Members need asset visibility and qualified introductions across the network
  \end{itemize}
  \vspace{0.5em}
  OSM-based layers refresh with the corridor; cadastral detail remains with planning authorities.
\end{frame}
