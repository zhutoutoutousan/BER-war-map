% --- Why BER+ ---
\begin{frame}{Why should BER+ care?}
  BER+ faces \textbf{coordination problems}, not only technology gaps:
  \begin{enumerate}
    \item Matching companies and opportunities
    \item Visibility of assets and spaces
    \item More value for members
    \item Accelerating regional development \& stakeholder alignment
  \end{enumerate}
  \vspace{0.5em}
  Our probe: a \textbf{war-room map} that makes corridor context shareable --- one screen for leadership briefings.
\end{frame}

% --- Why now ---
\begin{frame}{Why now? (12--24 months)}
  \begin{itemize}
    \item Grid congestion \& Netzanschluss delays stall gewerbe expansion \emph{today}
    \item FBB / FEW energy \& resilience targets are active compliance drivers
    \item SEGRO \textbf{Pilot-1} (2\,ha) anchors Phase I --- financeable proof unit, not ``a new city''
    \item Members asked for asset visibility \& matching (BER+ workshop, April 2026)
  \end{itemize}
  \vspace{0.5em}
  \textbf{Horizon:} validate indicative OSM + member links quarterly --- not replace cadastral GIS.
\end{frame}
